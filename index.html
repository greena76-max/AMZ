Website par Products bhejne ke liye (GET Request) function doGet(e) { var sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName("Sheet1"); // Agar sheet ka naam kuch aur hai toh yahan change kar lein var rows = sheet.getDataRange().getValues(); var headers = rows[0]; var data = []; for (var i = 1; i < rows.length; i++) { var row = rows[i]; var obj = {}; for (var j = 0; j < headers.length; j++) { if (headers[j]) { obj[headers[j].toString().toLowerCase().trim()] = row[j]; } } data.push(obj); } return ContentService.createTextOutput(JSON.stringify(data)) .setMimeType(ContentService.MimeType.JSON); } // 2. Website se Order save karne ke liye (POST Request) function doPost(e) { try { var sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName("Sheet1"); // Wahi sheet jahan orders store honge var data = JSON.parse(e.postData.contents); // Unique Order ID aur Date generate karna var orderId = "HY-" + Math.floor(100000 + Math.random() * 900000); var currentDate = new Date(); // Columns ke mutabiq data append hoga: // A: Date | B: Customer Name | C: Phone | D: Address | E: Product Name | F: Price | G: Payment Method | H: Order ID | I: Order Status sheet.appendRow([ currentDate, data.custName, data.custPhone, data.custAddress, data.name, data.price, "Cash on Delivery", // Payment Method orderId, "Pending" // Order Status ]); return ContentService.createTextOutput(JSON.stringify({status: "success", message: "Order placed successfully!"})) .setMimeType(ContentService.MimeType.JSON); } catch (error) { return ContentService.createTextOutput(JSON.stringify({status: "error", message: error.toString()})) .setMimeType(ContentService.MimeType.JSON); } }
